        {
            \ifthenelse{\(\NOT \i=3\)}
            {
            \draw[line width=1, grayed, |-|]
            (\eps+5+2*\x/5+\eps*\x, 0.3) -- (\eps+5+2*\x/5+0.4+\eps*\x, 0.3);

    }

    % the vertex jobs of v
